const SPREADSHEET_ID = '1_ITPoesKYY8NMjwwcextvIlCyG-PNsvCO2i4LFklr3I'; const SHEET_NAME = 'Sheet1'; const PROCESS_OPTIONS = [ 'Customer Contact', 'Customer Consent Obtained', 'Documents Awaited', 'Documents Received', 'Claim / Reversal Request Submitted', 'Reversal Under Process', 'Reversal Completed', 'Account Holder Not Traceable', 'Customer Not Interested', 'Other' ]; const STATUS_OPTIONS = [ 'Pending', 'In Progress', 'Completed', 'Not Eligible', 'Rejected', 'Closed' ]; function doGet() { return json_({ success: true, message: 'DEAF API is running' }); } function doPost(e) { try { const body = e && e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {}; const action = String(body.action || '').toLowerCase(); if (action === 'list') { return json_(listAccounts_()); } if (action === 'update') { return json_(updateAccount_(body)); } return json_({ success: false, error: 'Unknown API action: ' + action }); } catch (err) { return json_({ success: false, error: String(err.message || err) }); } } function json_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function listAccounts_() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sh = ss.getSheetByName(SHEET_NAME); if (!sh) { throw new Error('Sheet1 not found'); } const data = sh.getDataRange().getDisplayValues(); if (!data.length) { return { success: true, headers: [], records: [] }; } const headers = data[0].map(h => String(h).trim()); const col = {}; headers.forEach((h, i) => { col[h.toUpperCase()] = i; }); ['SOL ID', 'ACCOUNT NO'].forEach(h => { if (col[h] === undefined) { throw new Error('Missing column: ' + h); } }); const records = []; for (let i = 1; i < data.length; i++) { const row = data[i]; if (!row.some(v => String(v).trim() !== '')) { continue; } const item = { rowNumber: i + 1, values: row }; headers.forEach((h, j) => { item[h] = row[j]; }); item.solId = row[col['SOL ID']]; item.accountNo = row[col['ACCOUNT NO']]; item.process = col['PROCESS'] !== undefined ? row[col['PROCESS']] : ''; item.status = col['STATUS'] !== undefined ? row[col['STATUS']] : ''; item.remark = col['REMARK'] !== undefined ? row[col['REMARK']] : ''; records.push(item); } return { success: true, headers: headers, records: records, processOptions: PROCESS_OPTIONS, statusOptions: STATUS_OPTIONS }; } function updateAccount_(body) { const rowNumber = Number(body.rowNumber); if (!Number.isInteger(rowNumber) || rowNumber < 2) { throw new Error('Invalid spreadsheet row number'); } if (!PROCESS_OPTIONS.includes(String(body.process || ''))) { throw new Error('Invalid PROCESS value'); } if (!STATUS_OPTIONS.includes(String(body.status || ''))) { throw new Error('Invalid STATUS value'); } const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sh = ss.getSheetByName(SHEET_NAME); if (!sh) throw new Error('Sheet1 not found'); if (rowNumber > sh.getLastRow()) { throw new Error('Spreadsheet row does not exist'); } const headers = sh .getRange(1, 1, 1, sh.getLastColumn()) .getDisplayValues()[0] .map(h => String(h).trim().toUpperCase()); const col = {}; headers.forEach((h, i) => col[h] = i + 1); ['PROCESS', 'STATUS', 'REMARK'].forEach(h => { if (!col[h]) { throw new Error('Add the column ' + h + ' to row 1 first'); } }); sh.getRange(rowNumber, col['PROCESS']) .setValue(body.process); sh.getRange(rowNumber, col['STATUS']) .setValue(body.status); sh.getRange(rowNumber, col['REMARK']) .setValue(String(body.remark || '')); SpreadsheetApp.flush(); return { success: true, message: 'Account updated successfully' }; }
